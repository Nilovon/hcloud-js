```typescript
import { HCloudClient } from "@nilovonjs/hcloud-js";

const client = new HCloudClient();
```

## Provision a server

```typescript
const key =
  (await client.sshKeys.getByName("deploy")) ??
  (await client.sshKeys.create({ name: "deploy", public_key: process.env.DEPLOY_PUBLIC_KEY! })).ssh_key;

const { server } = await client.servers.createAndWait({
  name: "web-1",
  server_type: "cx22",
  image: "ubuntu-24.04",
  location: "nbg1",
  ssh_keys: [key.id],
  labels: { env: "production", role: "web" },
  user_data: "#cloud-config\npackages:\n  - nginx\n",
});

console.log(`ssh root@${server.public_net.ipv4?.ip}`);
```

## Idempotent "ensure" helpers

`getByName()` returns `undefined` instead of throwing, which makes
create-if-missing flows short:

```typescript
async function ensureNetwork(name: string) {
  const existing = await client.networks.getByName(name);
  if (existing) return existing;

  const { network } = await client.networks.create({
    name,
    ip_range: "10.0.0.0/16",
    subnets: [{ type: "cloud", ip_range: "10.0.1.0/24", network_zone: "eu-central" }],
  });
  return network;
}
```

## Lock down a server with a firewall

```typescript
const { firewall, actions } = await client.firewalls.create({
  name: "web",
  rules: [
    { direction: "in", protocol: "tcp", port: "22", source_ips: ["203.0.113.7/32"], description: "SSH from office" },
    { direction: "in", protocol: "tcp", port: "80", source_ips: ["0.0.0.0/0", "::/0"] },
    { direction: "in", protocol: "tcp", port: "443", source_ips: ["0.0.0.0/0", "::/0"] },
  ],
  apply_to: [{ type: "label_selector", label_selector: { selector: "role=web" } }],
});

await client.actions.waitForAll(actions);
```

## Attach and resize a volume

```typescript
const created = await client.volumes.create({
  name: "data",
  size: 50,
  server: server.id,
  format: "ext4",
  automount: true,
});
await client.actions.waitForAll(created);

await client.actions.waitFor(await client.volumes.resize(created.volume.id, { size: 100 }));
```

## Snapshot before a risky change

```typescript
const snapshot = await client.servers.createImage(server.id, {
  type: "snapshot",
  description: `pre-upgrade-${new Date().toISOString()}`,
  labels: { purpose: "pre-upgrade" },
});
await client.actions.waitFor(snapshot);

try {
  await runUpgrade(server);
} catch (error) {
  // Roll back
  await client.actions.waitFor(await client.servers.rebuild(server.id, { image: snapshot.image.id }));
  throw error;
}
```

## Rescale a server

Changing the server type requires the server to be powered off:

```typescript
await client.actions.waitFor(await client.servers.shutdown(server.id));
await client.servers.waitForStatus(server.id, "off", { timeout: 120_000 });

await client.actions.waitFor(
  await client.servers.changeType(server.id, { server_type: "cx32", upgrade_disk: false }),
);
await client.actions.waitFor(await client.servers.powerOn(server.id));
```

## Clean up by label

```typescript
import { labelSelector } from "@nilovonjs/hcloud-js";

const selector = labelSelector({ "created-by": "ci", keep: false });

const stale = await client.servers.listAll({ label_selector: selector });
const deletions = await Promise.all(stale.map((s) => client.servers.delete(s.id)));

await client.actions.waitForAll(deletions.map((deletion) => deletion.action));
console.log(`Deleted ${stale.length} servers`);
```

## Manage DNS records

```typescript
// Point www.example.com to a server
await client.actions.waitFor(
  await client.dns.setRRSetRecords("example.com", "www", "A", {
    records: [{ value: server.public_net.ipv4!.ip }],
  }),
);

// List every record of a zone
for await (const rrset of client.dns.iterateRRSets("example.com")) {
  console.log(rrset.name, rrset.type, rrset.records.map((record) => record.value));
}
```

## Monthly cost overview

```typescript
const servers = await client.servers.listAll();

let total = 0;
for (const server of servers) {
  const location = server.location?.name ?? server.datacenter?.location.name;
  const price = server.server_type.prices.find((p) => p.location === location);
  total += Number(price?.price_monthly.gross ?? 0);
}

console.log(`${servers.length} servers, ~${total.toFixed(2)} EUR per month`);
```

## Server metrics

```typescript
import { parseTimeSeries } from "@nilovonjs/hcloud-js";

const { metrics } = await client.servers.getMetrics(server.id, {
  type: ["cpu", "network"],
  start: new Date(Date.now() - 6 * 60 * 60 * 1000),
  end: new Date(),
});

const series = parseTimeSeries(metrics.time_series);
const cpu = series.cpu ?? [];
const average = cpu.reduce((sum, point) => sum + point.value, 0) / (cpu.length || 1);
console.log(`Average CPU over 6h: ${average.toFixed(1)}%`);
```

## Graceful shutdown of a long task

```typescript
const controller = new AbortController();
process.once("SIGINT", () => controller.abort());

try {
  for await (const image of client.images.iterate({ type: "backup" }, { signal: controller.signal })) {
    await archive(image);
  }
} catch (error) {
  if (error instanceof HCloudAbortError) console.log("Interrupted");
  else throw error;
}
```

## Testing your own code

Inject a fake `fetch` to test code that uses the SDK without network access:

```typescript
import { expect, test } from "vitest";
import { HCloudClient } from "@nilovonjs/hcloud-js";

test("counts running servers", async () => {
  const client = new HCloudClient({
    token: "test",
    responseValidation: "off",
    fetch: async (url) => {
      expect(new URL(url).pathname).toBe("/v1/servers");
      return Response.json({ servers: [{ id: 1, status: "running" }] });
    },
  });

  expect(await countRunningServers(client)).toBe(1);
});
```
